import pandas as pd
from sklearn.datasets import load_iris

# Load Iris dataset
iris = load_iris()
df = pd.DataFrame(iris.data, columns=iris.feature_names)
df['species'] = iris.target
df['species'] = df['species'].map(dict(enumerate(iris.target_names)))

# Group by species
species_grp = df.groupby('species')

# Subsets
setosa = species_grp.get_group('setosa')
versicolor = species_grp.get_group('versicolor')
virginica = species_grp.get_group('virginica')

# Group statistics
species_grp.mean(numeric_only=True)
species_grp.describe()
species_grp.groups

# Mean, median, mode for setosa sepal length
setosa['sepal length (cm)'].mean()
setosa['sepal length (cm)'].median()
setosa['sepal length (cm)'].mode()[0]
